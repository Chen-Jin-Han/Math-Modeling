    dist[start] = 0.0
    pq = [(0.0, start)]

    while pq:
        d, u = heapq.heappop(pq)
        if u == end:
            break
        for v, length, tid in graph[u]:
            if tid in water_time:
                speed = 2.0 if d >= water_time[tid] else 4.0
            else:
                speed = 4.0
            nd = d + length / speed
            if nd < dist[v]:
                dist[v] = nd
                prev[v] = u
                heapq.heappush(pq, (nd, v))

    path, node = [], end
    while node is not None:
        path.append(node)
        node = prev[node]
    path.reverse()
    return dist[end], path
\end{filecontents*}
